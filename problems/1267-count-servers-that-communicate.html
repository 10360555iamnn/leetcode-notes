1267. Count Servers that Communicate

統計參與通信的伺服器 Medium

LeetCode 1267 ↗在 YouTube 觀看 ↗

重點

先數每一列、每一行各有幾台伺服器；一台伺服器只要所在的列或行超過一台，就能通訊。也介紹看成圖、DFS、並查集的另解。

計數矩陣

C++ 解答 下載 .cpp

class Solution {
public:
    int countServers(vector<vector<int>>& grid) {
        int m = grid.size(), n = grid[0].size();
        vector<int> rowCnt(m, 0), colCnt(n, 0);
        for (int i = 0; i < m; i++)
            for (int j = 0; j < n; j++)
                if (grid[i][j] == 1) {
                    rowCnt[i]++;
                    colCnt[j]++;
                }
        int ans = 0;
        for (int i = 0; i < m; i++)
            for (int j = 0; j < n; j++)
                if (grid[i][j] == 1 && (rowCnt[i] > 1 || colCnt[j] > 1))
                    ans++;
        return ans;
    }
};
