重點整理

一段子陣列的花費 = 每個數墊到前綴最大值要加多少。花費有兩個單調性，所以由右往左移動左端點、用滑動視窗收右端；單調雙端佇列把「補到的高度」分塊存成（高度, 個數），加入和移除都是均攤 O(1)，總共 O(n)，記得用 long long。

滑動視窗單調佇列

C++ 解答 下載 .cpp

C++ · 滑動視窗 + 單調雙端佇列
class Solution {
public:
    long long countNonDecreasingSubarrays(vector<int>& nums, int k) {
        int n = nums.size();
        deque<pair<int, int>> dq;          // (補到的高度, 個數)，front = 視窗左端
        long long cost = 0, ans = 0;
        int r = n - 1;
        for (int l = n - 1; l >= 0; l--) {
            int cnt = 1;
            while (!dq.empty() && dq.front().first <= nums[l]) {
                cost += (long long)(nums[l] - dq.front().first) * dq.front().second;
                cnt += dq.front().second;
                dq.pop_front();
            }
            dq.push_front({nums[l], cnt});
            while (cost > k) {
                cost -= dq.back().first - nums[r];
                r--;
                if (--dq.back().second == 0) dq.pop_back();
            }
            ans += r - l + 1;
        }
        return ans;
    }
};
